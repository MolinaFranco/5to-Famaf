```
5to-Famaf/

│   └── herramientas/    # Alloy, LTSA, Infer, alloytools

│   ├── guias/           # Guías de la materia
│   ├── lab_2020/        # Labs (Lab1, Lab2, Lab3)
│   ├── LenguajesYCompiladoresLab/  # Repo del lab
│   └── material/        # Libros de referencia
│
├── Neuro/               # Redes Neuronales
│   ├── practicos/       # TPs (TP1, TP2 + imágenes)
│   └── material/        # Libro de Tom Mitchell
│
├── TADS/                # TADS

    ├── presentacion/    # Presentaciones (.pdf, .pptx)
    ├── papers/          # Papers de referencia
    ├── libros/          # Libros (master-algorithm, origins-of-neuroscience)
    ├── diagramas/       # Diagramas UML y drawio
    ├── administrativo/  # Carátula, plan, nota de inscripción, RHCD, CV

    ├── benchmarks/      # Benchmarks
    ├── pycloude/        # Helpers
```
